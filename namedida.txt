import Foundation
import Network
import NearbyInteraction

class GameNetworkService: NSObject, NISessionDelegate {
    private let serviceType = "_undercover._tcp"
    private let serviceDomain = "local."
    private let queue = DispatchQueue(label: "com.game.networkQueue")
    
    private var listener: NWListener?
    private(set) var connectedClients: [NWConnection] = []
    
    // Gerenciadores do NearbyInteraction
    private var activeSessions: [NISession: NWConnection] = [:]
    private var guestSession: NISession?
    
    private var browser: NWBrowser?
    private var hostConnection: NWConnection?
    
    var onClientsUpdated: ((Int) -> Void)?
    var onPayloadReceived: ((GamePayload) -> Void)?
    var onConnectionRejected: (() -> Void)? // Novo aviso para expulsão
    
    func startHosting() {
        do {
            listener = try NWListener(using: .tcp)
            listener?.service = NWListener.Service(name: "Sala Undercover", type: serviceType)
            listener?.newConnectionHandler = { [weak self] newConnection in
                self?.handleNewClientConnection(newConnection)
            }
            listener?.start(queue: queue)
        } catch {
            print("Erro ao iniciar o Host: \(error)")
        }
    }
    
    private func handleNewClientConnection(_ connection: NWConnection) {
        connection.stateUpdateHandler = { [weak self] state in
            guard let self = self else { return }
            switch state {
            case .ready:
                self.receiveData(on: connection)
                self.startNearbyHandshake(for: connection)
            case .failed, .cancelled:
                self.removeConnection(connection)
            default:
                break
            }
        }
        connection.start(queue: queue)
    }
    
    func startBrowsing() {
        browser = NWBrowser(for: .bonjour(type: serviceType, domain: serviceDomain), using: .tcp)
        browser?.browseResultsChangedHandler = { [weak self] results, _ in
            guard let self = self, let result = results.first else { return }
            self.connectToHost(endpoint: result.endpoint)
            self.browser?.cancel()
        }
        browser?.start(queue: queue)
    }
    
    private func connectToHost(endpoint: NWEndpoint) {
        let connection = NWConnection(to: endpoint, using: .tcp)
        connection.stateUpdateHandler = { [weak self] state in
            guard let self = self else { return }
            if case .ready = state {
                self.hostConnection = connection
                self.receiveData(on: connection)
                
                // Guest inicia a sua sessão para o Host
                self.guestSession = NISession()
                self.guestSession?.delegate = self
                self.guestSession?.delegateQueue = self.queue
                
                self.sendLocalToken(from: self.guestSession, to: connection)
            }
        }
        connection.start(queue: queue)
    }
    
    // MARK: - Lógica do Nearby Interaction
    
    private func startNearbyHandshake(for connection: NWConnection) {
        let session = NISession()
        session.delegate = self
        session.delegateQueue = queue
        activeSessions[session] = connection
        
        sendLocalToken(from: session, to: connection)
    }
    
    private func sendLocalToken(from session: NISession?, to connection: NWConnection) {
        guard let token = session?.discoveryToken,
              let tokenData = try? NSKeyedArchiver.archivedData(withRootObject: token, requiringSecureCoding: true) else { return }
        
        let message = NetworkMessage(tokenData: tokenData, gamePayload: nil)
        sendMessage(message, to: connection)
    }
    
    func session(_ session: NISession, didUpdate nearbyObjects: [NINearbyObject]) {
        guard let peer = nearbyObjects.first, let distance = peer.distance else { return }
        
        let connection = activeSessions[session] ?? hostConnection
        
        // VALIDAÇÃO DOS 5 METROS
        if distance <= 5.0 {
            if let conn = connection, conn !== hostConnection {
                // É o Host validando um Guest
                if !connectedClients.contains(where: { $0 === conn }) {
                    connectedClients.append(conn)
                    notifyClientsUpdate()
                }
            } else if connection === hostConnection {
                // É o Guest validando proximidade com o Host
                DispatchQueue.main.async { self.onClientsUpdated?(0) }
            }
        } else {
            // Fora do raio permitido: Derruba a conexão imediatamente
            connection?.cancel()
            session.invalidate()
            
            if let conn = connection {
                removeConnection(conn)
            }
            
            DispatchQueue.main.async {
                self.onConnectionRejected?()
            }
        }
    }
    
    // MARK: - Envio e Recebimento de Dados
    
    func sendMessage(_ message: NetworkMessage, to connection: NWConnection) {
        do {
            let data = try JSONEncoder().encode(message)
            connection.send(content: data, completion: .contentProcessed({ error in
                if let error = error { print("Erro ao enviar: \(error)") }
            }))
        } catch {
            print("Erro ao codificar mensagem: \(error)")
        }
    }
    
    private func receiveData(on connection: NWConnection) {
        connection.receive(minimumIncompleteLength: 1, maximumLength: 65536) { [weak self] data, _, isComplete, error in
            if let data = data, !data.isEmpty {
                if let message = try? JSONDecoder().decode(NetworkMessage.self, from: data) {
                    self?.processMessage(message, from: connection)
                }
            }
            if error == nil && !isComplete {
                self?.receiveData(on: connection)
            }
        }
    }
    
    private func processMessage(_ message: NetworkMessage, from connection: NWConnection) {
        // Se recebeu um Token do Nearby, configura e roda a sessão
        if let tokenData = message.tokenData {
            if let peerToken = try? NSKeyedUnarchiver.unarchivedObject(ofClass: NIDiscoveryToken.self, from: tokenData) {
                let config = NINearbyPeerConfiguration(peerToken: peerToken)
                
                if let session = activeSessions.first(where: { $0.value === connection })?.key {
                    session.run(config) // Host rodando
                } else if let guestSession = guestSession, connection === hostConnection {
                    guestSession.run(config) // Guest rodando
                }
            }
        }
        
        // Se recebeu dados do Jogo (A palavra)
        if let gamePayload = message.gamePayload {
            DispatchQueue.main.async {
                self.onPayloadReceived?(gamePayload)
            }
        }
    }
    
    private func removeConnection(_ connection: NWConnection) {
        connectedClients.removeAll(where: { $0 === connection })
        if let session = activeSessions.first(where: { $0.value === connection })?.key {
            session.invalidate()
            activeSessions.removeValue(forKey: session)
        }
        notifyClientsUpdate()
    }
    
    private func notifyClientsUpdate() {
        DispatchQueue.main.async {
            self.onClientsUpdated?(self.connectedClients.count)
        }
    }
    
    func stopAll() {
        listener?.cancel()
        listener = nil
        browser?.cancel()
        browser = nil
        hostConnection?.cancel()
        hostConnection = nil
        
        guestSession?.invalidate()
        guestSession = nil
        
        for session in activeSessions.keys { session.invalidate() }
        activeSessions.removeAll()
        
        for client in connectedClients { client.cancel() }
        connectedClients.removeAll()
    }
}

import SwiftUI
import Observation
import Network

@Observable
class GameViewModel {
    private var networkService = GameNetworkService()
    
    var isHost: Bool = false
    var isConnectedAsGuest: Bool = false
    var playersInRoom: Int = 0
    
    var wordToShow: String? = nil
    var isImpostor: Bool = false
    
    private let wordsPool = ["Flor", "MacBook", "Cachorro", "Futebol", "Teclado", "Scooby-Doo"]
    
    init() {
        setupNetworkCallbacks()
    }
    
    private func setupNetworkCallbacks() {
        networkService.onClientsUpdated = { [weak self] count in
            self?.playersInRoom = count
            self?.isConnectedAsGuest = true
        }
        
        networkService.onPayloadReceived = { [weak self] payload in
            self?.wordToShow = payload.word
            self?.isImpostor = payload.isImpostor
        }
    }
    
    func createRoom() {
        isHost = true
        networkService.startHosting()
    }
    
    func joinRoom() {
        isHost = false
        networkService.startBrowsing()
    }
    
    func startGame() {
            guard let randomWord = wordsPool.randomElement() else { return }
            
            let clients = networkService.connectedClients
            let totalPlayers = clients.count + 1
            
            let impostorIndex = Int.random(in: 0..<totalPlayers)
            
            self.isImpostor = (impostorIndex == 0)
            self.wordToShow = self.isImpostor ? "Você é o Impostor!" : randomWord
            
            for (index, clientConnection) in clients.enumerated() {
                let isThisClientImpostor = ((index + 1) == impostorIndex)
                let wordToSend = isThisClientImpostor ? "Você é o Impostor!" : randomWord
                let gamePayload = GamePayload(word: wordToSend, isImpostor: isThisClientImpostor)
                
                // Envia utilizando a nova estrutura de mensagem
                let message = NetworkMessage(tokenData: nil, gamePayload: gamePayload)
                networkService.sendMessage(message, to: clientConnection)
            }
        }
    
    //encerra o jogo atual
    func disconnect() {
        networkService.stopAll()
    }
}

import SwiftUI

struct ContentView: View {
    @State private var viewModel = GameViewModel()
    @State private var mode: AppMode = .menu
    
    enum AppMode { case menu, hosting, joining }
    
    var body: some View {
        VStack {
            if viewModel.wordToShow != nil {
                roleRevealView
            } else {
                switch mode {
                case .menu:
                    menuView
                case .hosting:
                    hostView
                case .joining:
                    clientView
                }
            }
        }
        .padding()
        .animation(.easeInOut, value: mode)
        .animation(.easeInOut, value: viewModel.wordToShow)
    }
    
    private var menuView: some View {
        VStack(spacing: 20) {
            
            Button("Criar Partida") {
                mode = .hosting
                viewModel.createRoom()
            }
            .buttonStyle(.borderedProminent)
            .controlSize(.large)
            
            Button("Entrar em uma partida") {
                mode = .joining
                viewModel.joinRoom()
            }
            .buttonStyle(.borderedProminent)
            .tint(.green)
            .controlSize(.large)
        }
    }
    
    private var hostView: some View {
        VStack(spacing: 20) {
            Text("Total de jogadores: \(viewModel.playersInRoom)")
                .font(.title2)
            
            if viewModel.playersInRoom > 0 {
                Button("Começar Partida") {
                    viewModel.startGame()
                }
                .buttonStyle(.borderedProminent)
                .tint(.blue)
                .controlSize(.large)
            }
                        
            Button("Cancelar") {
                resetGame()
            }
            .buttonStyle(.bordered)
            .tint(.red)
            .padding(.bottom, 10)
        }
    }
    
    private var clientView: some View {
        VStack(spacing: 20) {
                        
            Text("Esperando partida começar")
                .font(.title2)
                .multilineTextAlignment(.center)
                        
            Button("Sair") {
                resetGame()
            }
            .buttonStyle(.bordered)
            .tint(.red)
            .padding(.bottom, 10)
        }
    }
    
    private var roleRevealView: some View {
        VStack(spacing: 30) {
            Spacer()
            
            if viewModel.isImpostor {
                Text("VOCÊ É O IMPOSTOR")
                    .font(.largeTitle).bold()
                    .foregroundColor(.red)
                    .multilineTextAlignment(.center)
            } else {
                Text("A palavra secreta é:")
                    .font(.title2)
                
                Text(viewModel.wordToShow ?? "")
                    .font(.system(size: 40, weight: .bold))
                    .foregroundColor(.green)
                    .multilineTextAlignment(.center)
            }
            
            Spacer()
            
            Button("Sair da Partida") {
                resetGame()
            }
            .buttonStyle(.bordered)
            .tint(.red)
            .padding(.bottom, 10)
        }
    }
    
    private func resetGame() {
        
        //derruba a rede ativa para não bugar a próxima partida
        viewModel.disconnect()
        
        //cria um modelo limpo
        viewModel = GameViewModel()
        
        mode = .menu
    }
}
